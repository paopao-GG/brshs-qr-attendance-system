# UI
PySide6>=6.8            # LGPL binding. aarch64 wheel needs glibc >= 2.39 (Pi OS Trixie)
qtpy>=2.4               # binding abstraction so PySide6/PyQt6 is a config change

# SMS + QR
# SIM800C over USB serial. Pure Python, so unlike PySide6 and zxing-cpp it carries no
# glibc or architecture constraint on the Pi.
pyserial>=3.5
qrcode[pil]>=7.4
Pillow>=10.0

# Webcam QR decoding. 1 MB wheel; the aarch64 build needs only glibc >= 2.26,
# a looser bar than PySide6 itself. Deliberately not OpenCV (~90 MB, and its
# bundled Qt plugins break PySide6 on Linux) and not pyzbar (needs libzbar0).
zxing-cpp>=2.2

# Analytics
numpy>=1.26
statsmodels>=0.14       # linear regression + Durbin-Watson
scikit-learn>=1.4       # logistic regression, precision/recall/AUC

# Reports
# XLSX register export. Pure Python, so no glibc or architecture constraint on the Pi.
# Present here only as a pandas dependency until now, which is not a declaration.
openpyxl>=3.1

# Security
argon2-cffi>=23.1

# Transitive, pinned deliberately. statsmodels 0.15 pulls in formulaic, which declares
# `wrapt>=1.17.0rc1` on Python 3.13 -- and because that specifier NAMES a pre-release,
# PEP 440 makes every pre-release eligible for it. pip then resolved the newest match
# overall, wrapt 2.4.0rc5, in preference to stable 2.3.0. A release candidate is not
# what a school gate station should be running; stable satisfies the constraint fine.
wrapt>=2.3.0,<3
